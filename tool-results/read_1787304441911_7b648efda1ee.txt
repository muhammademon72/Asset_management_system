  1240→    data.filter(function(d){return d.status==='Rejected';}).length,

  1251→    ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill();
  1252→    startAngle += sliceAngle;
  1253→  }
  1254→  ctx.beginPath(); ctx.arc(cx, cy, Math.max(r * 0.55, 1), 0, Math.PI * 2);
