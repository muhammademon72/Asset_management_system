   374→      html += '<div class="submenu ' + (isOpen ? 'open' : '') + '">';
   375→      item.children.forEach(function(ch) {
   376→        html += '<div class="nav-item ' + (currentPage === ch.id ? 'active' : '') + '" onclick="event.stopPropagation();navigate(\'' + ch.id + '\')">';

   557→function initDashboardChart() {
   558→  var counters = document.querySelectorAll('.counter');
   559→  for (var i = 0; i < counters.length; i++) {
   560→    (function(el) {
   561→      var target = parseInt(el.getAttribute('data-target'));
   562→      var prefix = el.getAttribute('data-prefix') || '';
   563→      var current = 0;
   564→      var step = Math.max(1, Math.floor(target / 40));
   565→      var interval = setInterval(function() {
   566→        current += step;
   567→        if (current >= target) { current = target; clearInterval(interval); }
   568→        el.textContent = prefix + current.toLocaleString('en-IN');
   569→      }, 25);
   570→    })(counters[i]);
   571→  }
   572→
   573→  var canvas = document.getElementById('dashChart');
   574→  if (!canvas) return;
   575→  var ctx = canvas.getContext('2d');
   576→  var W = canvas.width, H = canvas.height;
   577→  var data = [850000, 390000, 135000, 220000, 0, 0];
   578→  var labels = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'];
   579→  var maxVal = Math.max.apply(null, data) || 1;
   580→  var barW = 50, gap = (W - data.length * barW) / (data.length + 1);
   581→  var chartH = H - 40;
   582→  var progress = 0;
   583→
   584→  function drawBars() {
   585→    ctx.clearRect(0, 0, W, H);
   586→    ctx.strokeStyle = '#e5e7eb'; ctx.lineWidth = 1;
   587→    for (var i = 0; i <= 4; i++) {
   588→      var y = 10 + (chartH / 4) * i;
   589→      ctx.beginPath(); ctx.moveTo(40, y); ctx.lineTo(W - 10, y); ctx.stroke();
   590→      ctx.fillStyle = '#9ca3af'; ctx.font = '11px Outfit'; ctx.textAlign = 'right';
   591→      ctx.fillText(fmt(Math.round(maxVal - (maxVal / 4) * i)), 38, y + 4);
   592→    }
   593→    for (var j = 0; j < data.length; j++) {
   594→      var x = gap + j * (barW + gap);
   595→      var barH = (data[j] / maxVal) * chartH * Math.min(progress, 1);
   596→      var by = 10 + chartH - barH;
   597→      if (barH > 1) {
   598→        var grad = ctx.createLinearGradient(x, by, x, 10 + chartH);
   599→        grad.addColorStop(0, '#2d8a4e');
   600→        grad.addColorStop(1, '#86efac');
   601→        ctx.fillStyle = grad;
   602→        ctx.beginPath();
   603→        var r = Math.min(6, barH / 2);
   604→        ctx.moveTo(x + r, by);
   605→        ctx.arcTo(x + barW, by, x + barW, by + barH, r);
   606→        ctx.lineTo(x + barW, 10 + chartH);
   607→        ctx.lineTo(x, 10 + chartH);
   608→        ctx.arcTo(x, by, x + r, by, r);
   609→        ctx.fill();
   610→      }
   611→      ctx.fillStyle = '#6b7280'; ctx.font = '12px Outfit'; ctx.textAlign = 'center';
   612→      ctx.fillText(labels[j], x + barW / 2, H - 8);
   613→    }
   614→    if (progress < 1) { progress += 0.04; requestAnimationFrame(drawBars); }
   615→  }
   616→  drawBars();
   617→}
