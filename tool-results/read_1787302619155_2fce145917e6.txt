  1157→function drawReportPie() {
  1158→  var canvas = document.getElementById('reportPie');
  1159→  if (!canvas) return;
  1160→  var ctx = canvas.getContext('2d');
  1161→  var W = canvas.width, H = canvas.height;
  1162→  var type = currentPage.replace('report-', '');
  1163→  var map = { purchase: DB.purchases, delivery: DB.deliveries, adjustment: DB.adjustments, damage: DB.damages };
  1164→  var data = map[type];
  1165→  if (!data) return;
  1166→  var vals = [
  1167→    data.filter(function(d){return d.status==='Approved';}).length,
  1168→    data.filter(function(d){return d.status==='Pending';}).length,
  1169→    data.filter(function(d){return d.status==='Rejected';}).length,
  1170→  ];
  1171→  var colors = ['#2d8a4e', '#d97706', '#dc2626'];
  1172→  var lbls = ['Approved', 'Pending', 'Rejected'];
  1173→  var total = vals[0] + vals[1] + vals[2] || 1;
  1174→  var cx = W * 0.35, cy = H / 2, r = Math.min(cx, cy) - 20;
  1175→  var startAngle = -Math.PI / 2;
  1176→  for (var i = 0; i < vals.length; i++) {
  1177→    var sliceAngle = (vals[i] / total) * Math.PI * 2;
  1178→    ctx.beginPath(); ctx.moveTo(cx, cy);
  1179→    ctx.arc(cx, cy, Math.max(r, 1), startAngle, startAngle + sliceAngle);
  1180→    ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill();
  1181→    startAngle += sliceAngle;
  1182→  }
  1183→  ctx.beginPath(); ctx.arc(cx, cy, Math.max(r * 0.55, 1), 0, Math.PI * 2);
  1184→  ctx.fillStyle = '#ffffff'; ctx.fill();
  1185→  ctx.fillStyle = '#1a2e1a'; ctx.font = 'bold 22px Outfit'; ctx.textAlign = 'center';
  1186→  ctx.fillText(total, cx, cy + 4);
  1187→  ctx.font = '12px Outfit'; ctx.fillStyle = '#6b7c6b'; ctx.fillText('Total', cx, cy + 20);
  1188→  var lx = W * 0.7;
  1189→  for (var j = 0; j < lbls.length; j++) {
  1190→    var ly = 50 + j * 40;
  1191→    ctx.fillStyle = colors[j];
  1192→    ctx.beginPath(); ctx.rect(lx, ly, 16, 16); ctx.fill();
  1193→    ctx.fillStyle = '#1a2e1a'; ctx.font = '600 14px Outfit'; ctx.textAlign = 'left';
  1194→    ctx.fillText(lbls[j], lx + 24, ly + 13);
  1195→    ctx.fillStyle = '#6b7c6b'; ctx.font = '13px Outfit';
  1196→    ctx.fillText(vals[j] + ' items', lx + 24, ly + 30);
  1197→  }
  1198→}
