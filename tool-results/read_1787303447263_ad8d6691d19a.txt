   360→function renderNav() {
   361→  const c = document.getElementById('nav-container');
   362→  let html = '';
   363→  NAV.forEach(item => {
   364→    const hasChildren = item.children && item.children.length;
   365→    const isOpen = openMenus[item.id];
   366→    const isActive = !hasChildren && currentPage === item.id;
   367→    const isChildActive = hasChildren && item.children.some(function(ch){ return ch.id === currentPage; });
   368→    html += '<div class="nav-item ' + (isActive || isChildActive ? 'active' : '') + '" onclick="' + (hasChildren ? "toggleMenu('" + item.id + "')" : "navigate('" + item.id + "')") + '">';
   369→    html += '<i class="fas ' + item.icon + ' nav-icon"></i>';
   370→    html += '<span class="nav-text">' + item.label + '</span>';
   371→    if (hasChildren) html += '<i class="fas fa-chevron-right menu-arrow ' + (isOpen ? 'open' : '') + '"></i>';
   372→    html += '</div>';
   373→    if (hasChildren) {
   374→      html += '<div class="submenu ' + (isOpen ? 'open' : '') + '">';
   375→      item.children.forEach(function(ch) {
   376→        html += '<div class="nav-item ' + (currentPage === ch.id ? 'active' : '') + '" onclick="event.stopPropagation();navigate(\'' + ch.id + '\')">';
   377→        html += '<i class="fas fa-circle" style="font-size:5px;margin-right:12px;"></i>';
   378→        html += '<span class="nav-text">' + ch.label + '</span>';
   379→        html += '</div>';
   380→      });
   381→      html += '</div>';
   382→    }
   383→  });
   384→  c.innerHTML = html;
   385→}
   386→
   387→function toggleMenu(id) { openMenus[id] = !openMenus[id]; renderNav(); }

   389→function navigate(page) {
   390→  currentPage = page;
   391→  NAV.forEach(function(item) {
   392→    if (item.children) {
   393→      item.children.forEach(function(ch) {
   394→        if (ch.id === page) openMenus[item.id] = true;
   395→      });
   396→    }
   397→  });
   398→  renderNav();
   399→  renderPage();
   400→}

   557→function initDashboardChart() {
   558→  var counters = document.querySelectorAll('.counter');
   559→  for (var i = 0; i < counters.length; i++) {
   560→    (function(el) {
   561→      var target = parseInt(el.getAttribute('data-target'));
   562→      var prefix = el.getAttribute('data-prefix') || '';
   563→      var current = 0;
   564→      var step = Math.max(1, Math.floor(target / 40));
   565→      var interval = setInterval(function() {
   566→        current += step;
   567→        if (current >= target) { current = target; clearInterval(interval); }
   568→        el.textContent = prefix + current.toLocaleString('en-IN');
   569→      }, 25);
   570→    })(counters[i]);
   571→  }
   572→
   573→  var canvas = document.getElementById('dashChart');
   574→  if (!canvas) return;
   575→  var ctx = canvas.getContext('2d');
   576→  var W = canvas.width, H = canvas.height;
   577→  var data = [850000, 390000, 135000, 220000, 0, 0];
   578→  var labels = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'];
   579→  var maxVal = Math.max.apply(null, data) || 1;
   580→  var barW = 50, gap = (W - data.length * barW) / (data.length + 1);
   581→  var chartH = H - 40;
   582→  var progress = 0;
   583→
   584→  function drawBars() {
   585→    ctx.clearRect(0, 0, W, H);
   586→    ctx.strokeStyle = '#e5e7eb'; ctx.lineWidth = 1;
   587→    for (var i = 0; i <= 4; i++) {
   588→      var y = 10 + (chartH / 4) * i;
   589→      ctx.beginPath(); ctx.moveTo(40, y); ctx.lineTo(W - 10, y); ctx.stroke();
   590→      ctx.fillStyle = '#9ca3af'; ctx.font = '11px Outfit'; ctx.textAlign = 'right';
   591→      ctx.fillText(fmt(Math.round(maxVal - (maxVal / 4) * i)), 38, y + 4);
   592→    }
   593→    for (var j = 0; j < data.length; j++) {
   594→      var x = gap + j * (barW + gap);
   595→      var barH = (data[j] / maxVal) * chartH * Math.min(progress, 1);
   596→      var by = 10 + chartH - barH;
   597→      if (barH > 1) {
   598→        var grad = ctx.createLinearGradient(x, by, x, 10 + chartH);
   599→        grad.addColorStop(0, '#2d8a4e');
   600→        grad.addColorStop(1, '#86efac');
   601→        ctx.fillStyle = grad;
   602→        ctx.beginPath();
   603→        var r = Math.min(6, barH / 2);
   604→        ctx.moveTo(x + r, by);
   605→        ctx.arcTo(x + barW, by, x + barW, by + barH, r);
   606→        ctx.lineTo(x + barW, 10 + chartH);
   607→        ctx.lineTo(x, 10 + chartH);
   608→        ctx.arcTo(x, by, x + r, by, r);
   609→        ctx.fill();
   610→      }
   611→      ctx.fillStyle = '#6b7280'; ctx.font = '12px Outfit'; ctx.textAlign = 'center';
   612→      ctx.fillText(labels[j], x + barW / 2, H - 8);
   613→    }
   614→    if (progress < 1) { progress += 0.04; requestAnimationFrame(drawBars); }
   615→  }
   616→  drawBars();
   617→}

  1220→function drawReportPie() {
  1221→  var canvas = document.getElementById('reportPie');
  1222→  if (!canvas) return;
  1223→  var ctx = canvas.getContext('2d');
  1224→  var W = canvas.width, H = canvas.height;
  1225→  var type = currentPage.replace('report-', '');

  1229→  var vals = [
  1230→    data.filter(function(d){return d.status==='Approved';}).length,
  1231→    data.filter(function(d){return d.status==='Pending';}).length,
  1232→    data.filter(function(d){return d.status==='Rejected';}).length,
  1233→  ];
  1234→  var colors = ['#2d8a4e', '#d97706', '#dc2626'];
  1235→  var lbls = ['Approved', 'Pending', 'Rejected'];
  1236→  var total = vals[0] + vals[1] + vals[2] || 1;
  1237→  var cx = W * 0.35, cy = H / 2, r = Math.min(cx, cy) - 20;
  1238→  var startAngle = -Math.PI / 2;
  1239→  for (var i = 0; i < vals.length; i++) {
  1240→    var sliceAngle = (vals[i] / total) * Math.PI * 2;
  1241→    ctx.beginPath(); ctx.moveTo(cx, cy);
  1242→    ctx.arc(cx, cy, Math.max(r, 1), startAngle, startAngle + sliceAngle);
  1243→    ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill();
  1244→    startAngle += sliceAngle;
  1245→  }
  1246→  ctx.beginPath(); ctx.arc(cx, cy, Math.max(r * 0.55, 1), 0, Math.PI * 2);
  1247→  ctx.fillStyle = '#ffffff'; ctx.fill();
  1248→  ctx.fillStyle = '#1a2e1a'; ctx.font = 'bold 22px Outfit'; ctx.textAlign = 'center';
  1249→  ctx.fillText(total, cx, cy + 4);
  1250→  ctx.font = '12px Outfit'; ctx.fillStyle = '#6b7c6b'; ctx.fillText('Total', cx, cy + 20);

  1253→    var ly = 50 + j * 40;

  1255→    ctx.beginPath(); ctx.rect(lx, ly, 16, 16); ctx.fill();
  1256→    ctx.fillStyle = '#1a2e1a'; ctx.font = '600 14px Outfit'; ctx.textAlign = 'left';

  1259→    ctx.fillText(vals[j] + ' items', lx + 24, ly + 30);
