  1213→function drawReportPie() {
  1214→  var canvas = document.getElementById('reportPie');
  1215→  if (!canvas) return;
  1216→  var ctx = canvas.getContext('2d');
  1217→  var W = canvas.width, H = canvas.height;

  1223→    data.filter(function(d){return d.status==='Approved';}).length,
  1224→    data.filter(function(d){return d.status==='Pending';}).length,
  1225→    data.filter(function(d){return d.status==='Rejected';}).length,
  1226→  ];
  1227→  var colors = ['#2d8a4e', '#d97706', '#dc2626'];
  1228→  var lbls = ['Approved', 'Pending', 'Rejected'];
  1229→  var total = vals[0] + vals[1] + vals[2] || 1;

  1231→  var startAngle = -Math.PI / 2;
  1232→  for (var i = 0; i < vals.length; i++) {
  1233→    var sliceAngle = (vals[i] / total) * Math.PI * 2;
  1234→    ctx.beginPath(); ctx.moveTo(cx, cy);
  1235→    ctx.arc(cx, cy, Math.max(r, 1), startAngle, startAngle + sliceAngle);
  1236→    ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill();
  1237→    startAngle += sliceAngle;
  1238→  }
  1239→  ctx.beginPath(); ctx.arc(cx, cy, Math.max(r * 0.55, 1), 0, Math.PI * 2);
  1240→  ctx.fillStyle = '#ffffff'; ctx.fill();
  1241→  ctx.fillStyle = '#1a2e1a'; ctx.font = 'bold 22px Outfit'; ctx.textAlign = 'center';
  1242→  ctx.fillText(total, cx, cy + 4);
  1243→  ctx.font = '12px Outfit'; ctx.fillStyle = '#6b7c6b'; ctx.fillText('Total', cx, cy + 20);
  1244→  var lx = W * 0.7;
  1245→  for (var j = 0; j < lbls.length; j++) {
  1246→    var ly = 50 + j * 40;
  1247→    ctx.fillStyle = colors[j];
  1248→    ctx.beginPath(); ctx.rect(lx, ly, 16, 16); ctx.fill();
  1249→    ctx.fillStyle = '#1a2e1a'; ctx.font = '600 14px Outfit'; ctx.textAlign = 'left';
