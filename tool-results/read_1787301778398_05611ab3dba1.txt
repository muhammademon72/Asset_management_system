  1160→function drawReportPie() {
  1161→  var canvas = document.getElementById('reportPie');
  1162→  if (!canvas) return;
  1163→  var ctx = canvas.getContext('2d');
  1164→  var W = canvas.width, H = canvas.height;
  1165→  var type = currentPage.replace('report-', '');
  1166→  var map = { purchase: DB.purchases, delivery: DB.deliveries, adjustment: DB.adjustments, damage: DB.damages };
  1167→  var data = map[type];
  1168→  if (!data) return;
  1169→  var vals = [
  1170→    data.filter(function(d){return d.status==='Approved';}).length,
  1171→    data.filter(function(d){return d.status==='Pending';}).length,
  1172→    data.filter(function(d){return d.status==='Rejected';}).length,
  1173→  ];
  1174→  var colors = ['#2d8a4e', '#d97706', '#dc2626'];
  1175→  var lbls = ['Approved', 'Pending', 'Rejected'];
  1176→  var total = vals[0] + vals[1] + vals[2] || 1;
  1177→  var cx = W * 0.35, cy = H / 2, r = Math.min(cx, cy) - 20;
  1178→  var startAngle = -Math.PI / 2;
  1179→  for (var i = 0; i < vals.length; i++) {
  1180→    var sliceAngle = (vals[i] / total) * Math.PI * 2;
  1181→    ctx.beginPath(); ctx.moveTo(cx, cy);
  1182→    ctx.arc(cx, cy, Math.max(r, 1), startAngle, startAngle + sliceAngle);
  1183→    ctx.closePath(); ctx.fillStyle = colors[i]; ctx.fill();
  1184→    startAngle += sliceAngle;
  1185→  }
  1186→  ctx.beginPath(); ctx.arc(cx, cy, Math.max(r * 0.55, 1), 0, Math.PI * 2);
  1187→  ctx.fillStyle = '#ffffff'; ctx.fill();
  1188→  ctx.fillStyle = '#1a2e1a'; ctx.font = 'bold 22px Outfit'; ctx.textAlign = 'center';
  1189→  ctx.fillText(total, cx, cy + 4);
  1190→  ctx.font = '12px Outfit'; ctx.fillStyle = '#6b7c6b'; ctx.fillText('Total', cx, cy + 20);
  1191→  var lx = W * 0.7;
  1192→  for (var j = 0; j < lbls.length; j++) {
  1193→    var ly = 50 + j * 40;
  1194→    ctx.fillStyle = colors[j];
  1195→    ctx.beginPath(); ctx.rect(lx, ly, 16, 16); ctx.fill();
  1196→    ctx.fillStyle = '#1a2e1a'; ctx.font = '600 14px Outfit'; ctx.textAlign = 'left';
  1197→    ctx.fillText(lbls[j], lx + 24, ly + 13);
  1198→    ctx.fillStyle = '#6b7c6b'; ctx.font = '13px Outfit';
  1199→    ctx.fillText(vals[j] + ' items', lx + 24, ly + 30);
  1200→  }
  1201→}
